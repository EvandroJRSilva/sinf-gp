\section{Introdução}
	
\begin{frame}{Introdução}
	\justifying
	\begin{itemize}
		\justifying
		\item<1-4> Um \alert<2>{\textbf{Gerente de Projetos}}, um \alert<3>{\textbf{\textit{Tech Lead}}} e um \alert<4>{\textbf{Engenheiro de Software}} são papéis vitais nos times de desenvolvimento.
			\begin{itemize}
				\justifying
				\item<2> Prioriza o gerenciamento do time, incluindo a contratação, dinâmicas de equipe, facilitação de comunicação, e garantir as entregas. Geralmente evitam a programação do dia-a-dia para focar em questões estratégicas.
				\item<3> Lidera pelo exemplo. Geralmente trabalham diretamente no código e são responsáveis por decisões técnicas.
				\item<4> São desenvolvedores profissionais que trabalham na construção de produtos sob liderança do \textit{Tech Lead} e do Gerente do projeto.
			\end{itemize}
		\item<5> O desafio para um Gerente de Projetos reside em \textbf{equilibrar as responsabilidades de gestão} com a \textbf{manutenção de um conhecimento técnico apurado}. Um bom Gerente \textbf{reconhece essas diferenças}, \textbf{colabora eficazmente} e \textbf{garante o bom funcionamento da equipe}, ao mesmo tempo que \textbf{promove um ambiente propício ao crescimento e à aprendizagem}.
	\end{itemize}
\end{frame}

\section{Gerenciamento de Projeto de Software}

\begin{frame}{Gerenciamento de Projeto de Software}
	\begin{itemize}
		\justifying
		\item É uma atividade de apoio da Engenharia de Software. Inicia antes de qualquer atividade técnica e prossegue ao longo da modelagem, construção e utilização do software.
	\end{itemize}
\end{frame}

\begin{frame}{Gerenciamento de projeto de software}
	\begin{itemize}
		\justifying
		\item Segundo Pressman e Maxim, o \alert<3>{gerenciamento} eficiente do desenvolvimento de software se concentra nos \textbf{4 Ps}: \alert<2-3>{Pessoas}, produto, processo e projeto.
		\begin{itemize}
			\justifying
			\item<2> Toda organização precisa \underline{aprimorar continuamente} sua habilidade de \textbf{atrair}, \textbf{desenvolver}, \textbf{motivar}, \textbf{organizar} e \textbf{reter} a força de trabalho necessária para atingir os objetivos estratégicos de seus negócios.
			\item<3> Devem ser organizadas em \textbf{equipes eficientes}, devem ser \textbf{motivadas} a realizar um \textbf{trabalho} de software \textbf{de alta qualidade} e devem ser \textbf{coordenadas para uma comunicação eficaz}.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Gerenciamento de projeto de software}
	\begin{itemize}
		\justifying
		\item Segundo Pressman e Maxim, o \alert<2>{gerenciamento} eficiente do desenvolvimento de software se concentra nos \textbf{4 Ps}: Pessoas, \alert<1-2>{produto}, processo e projeto.
		\begin{itemize}
			\justifying
			\item<1> Antes de se iniciar um plano de projeto, é interessante \textbf{estabelecer} os \textbf{objetivos} do produto e seu \textbf{escopo}, considerar as soluções alternativas e \textbf{identificar} as \textbf{restrições técnicas e de gerenciamento}.
			\item<2> Requisitos de produto devem ser comunicados do cliente ao desenvolvedor, decompostos em partes e posicionados para a execução pela equipe de software.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Gerenciamento de projeto de software}
	\begin{itemize}
		\justifying
		\item Segundo Pressman e Maxim, o \alert<2>{gerenciamento} eficiente do desenvolvimento de software se concentra nos \textbf{4 Ps}: Pessoas, produto, \alert<1-2>{processo} e projeto.
		\begin{itemize}
			\justifying
			\item<1> \textbf{Fornece a metodologia} por meio da qual pode ser \textbf{estabelecido} um \textbf{plano de projeto} abrangente para o desenvolvimento de software, além de \textbf{atividades de apoio}, como garantia de qualidade, gerenciamento de configuração, me-dições, etc.
			\item<2> O processo \textbf{deve ser adaptado} às \textbf{pessoas} e ao \textbf{problema}.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Gerenciamento de projeto de software}
	\begin{itemize}
		\justifying
		\item Segundo Pressman e Maxim, o \alert<2>{gerenciamento} eficiente do desenvolvimento de software se concentra nos \textbf{4 Ps}: Pessoas, produto, processo e \alert<1-2>{projeto}.
		\begin{itemize}
			\justifying
			\item<1> \textbf{Planejamento} e \textbf{controle} são a \textbf{única maneira} de \textbf{administrar a complexidade}. São muitos os softwares que sofrem grande atraso, ou mesmo não são concluídos.
			\item<2> O \textbf{projeto}  deve ser \textbf{organizado} de modo a \textbf{capacitar} a \textbf{equipe} de software a \textbf{ser bem-sucedida}.
		\end{itemize}
	\end{itemize}
\end{frame}

\subsection{O princípio $W^{5}$HH}

\begin{frame}{O princípio $W^{5}$HH}
	\begin{itemize}
		\justifying
		\item É uma abordagem voltada aos objetivos do projeto, marcos (pontos de referência) e cronogramas (agendas), responsabilidades, gerenciamento, abordagens técnicas e recursos necessários.
		\item O nome vem de uma série de perguntas em Inglês, as quais conduzem a uma definição das características-chave do projeto e do planejamento do projeto resultante.
	\end{itemize}
\end{frame}

\begin{frame}{O princípio $W^{5}$HH}
	\begin{description}
		\item[Why] \textbf{Por que o sistema está sendo desenvolvido?}
		\item[What] \textbf{O que será feito?}
		\item[When] \textbf{Quando será feito?}
		\item[Who] \textbf{Quem será o responsável por uma função?}
		\item[Where] \textbf{Onde se posicionam organizacionalmente?}
		\item[How] \textbf{Como será realizado o trabalho de forma técnica e gerencialmente?}
		\item[How much] \textbf{Quanto de cada recurso será necessário?}
	\end{description}
\end{frame}

\section{Modelos de Processo Prescritivo}

\begin{frame}{Modelos de Processo Prescritivo}
	\begin{block}{Modelos de processo prescritivo}
		\justifying
		Definem um \textbf{conjunto prescrito} de \underline{elementos de processo} e \underline{um fluxo de trabalho de processo \textbf{previsível}}.        
	\end{block}
	
	\begin{itemize}
		\justifying
		\item<2-> Um modelo concentra-se em \textbf{estruturar} e \textbf{ordenar} o desenvolvimento de software. As atividades e tarefas ocorrem \textbf{sequencialmente}, com \textbf{diretrizes de progresso} defini-das.
	\end{itemize}
\end{frame}

\subsection{Prototipação}

\begin{frame}{Prototipação}
	\begin{itemize}
		\justifying
		\item<1> Comumente o cliente define os \textbf{objetivos gerais} para o software, \textbf{mas não identifica os detalhes} (requisitos, funções, recursos, etc.)
		\item<1> Também é comum que \textbf{algum desenvolvedor não esteja seguro sobre uma funcionalidade}.
		\item<2> A prototipação auxilia os envolvidos a compreender melhor o que está sendo construído.
		\begin{itemize}
			\justifying
			\item Apesar de poder ser utilizado como um modelo de processo de software, \textbf{a prototipação é mais comumente utilizada como uma técnica a ser implementada no contexto dos demais modelos}, inclusive nas metodologias ágeis.
		\end{itemize}
	\end{itemize}
\end{frame}

\subsection{Modelo Evolucionário}

\begin{frame}{Modelo Evolucionário}
	\begin{itemize}
		\justifying
		\item<1> O software evolui ao longo do tempo. Conforme o desenvolvimento do projeto avança, os requisitos do negócio e do produto frequentemente mudam
		\begin{itemize}
			\justifying
			\item<1> Ou seja, seguir um planejamento em linha reta se torna inadequado.
		\end{itemize}
		\item<2> Prazos apertados (por pressões comerciais ou da concor-rência --- Bard/Gemini, a LLM do Google passou um aperreio) são muitas vezes comuns, o que impossibilita a conclusão de um software abrangente.
		\begin{itemize}
			\justifying
			\item<2> Possível solução: lançar uma versão limitada e depois uma versão mais refinada.
		\end{itemize}
		\item<3> Proposta: \textbf{Modelo Espiral}
		\begin{itemize}
			\justifying
			\item<3> Une a natureza iterativa da prototipação aos aspectos sistemáticos e controlados do modelo cascata.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Modelo Evolucionário}
	\begin{itemize}
		\justifying
		\item Com um modelo em espiral o software será desenvolvido em uma série de versões evolucionárias.
		\begin{itemize}
			\justifying
			\item<1> Nas primeiras iterações a versão do software pode ser um protótipo.
			\item<1> Nas iterações posteriores, são produzidas versões cada vez mais completas.
		\end{itemize}
		\item<2> \textbf{Marcos de pontos-âncora} --- uma combinação de artefatos e condições satisfeitas ao logo do trajeto da espiral --- são indicados para cada passagem evolucionária.
	\end{itemize}
\end{frame}

\subsection{Modelo Unificado}

\begin{frame}{Modelo Unificado}
	\begin{itemize}
		\justifying
		\item É uma \textbf{tentativa de aproveitar os melhores recursos e características} dos \textbf{modelos tradicionais}, mas caracterizando-os de modo a implementar muitos dos \textbf{melhores princípios do desenvolvimento ágil}.
		\begin{itemize}
			\justifying
			\item<1> Reconhece a \textbf{importância da comunicação com o cliente}.
			\item<2> Enfatiza o \textbf{importante papel da arquitetura} de software e ajuda o arquiteto a \textbf{manter o foco nas metas corretas}.
			\item<3> Sugere o \textbf{fluxo de processo iterativo e incremental}.
			\item<4> \textbf{A UML} (\textit{Unified Modeling Language}, ou Linguagem de Modelagem Unificada) \textbf{foi desenvolvida para apoiar o Modelo de Processo Unificado}.
		\end{itemize}
	\end{itemize}
\end{frame}
