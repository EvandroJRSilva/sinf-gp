\section{Documentação}

\begin{frame}{Documentação}
    \begin{itemize}
        \justifying
        \item Dependendo da empresa --- e quanto maior ela é, ou o projeto é --- é necessária a geração de uma documentação mais técnica de todo o projeto, a qual vai conter, normalmente, os seguintes conteúdos:
        	\begin{itemize}
        		\justifying
        		\item Documentação de arquitetura.
        		\item Documentação de processo.
        		\item Registro de decisões.
        	\end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Documentação}
	\begin{itemize}
		\justifying
		\item Além daqueles conteúdos, também podem estar presentes:
		\begin{itemize}
			\justifying
			\item Melhores práticas.
			\item Lições aprendidas.
		\end{itemize}
		\item Esses dois últimos conteúdos \textbf{podem ser registrados sem vinculação a um projeto específico}, uma vez que se trata de ganho de experiência.
		\item Outra forma bastante comum de registrá-los é através de documenta-ção geral interna (\textit{wikis}, convenções, documentação de \textit{framework} próprio, etc.) a ser consultada pelos desenvolvedores, cujos padrões são revisados e atualizados à medida em que os projetos são concluídos.
	\end{itemize}
\end{frame}

\subsubsection{Levantamento de requisitos - revisão}

\begin{frame}{Levantamento de requisitos - revisão}
	\begin{itemize}
		\justifying
		\item \alert<2>{\textbf{Requisitos funcionais}}: funcionalidades elencadas pelo cliente, ou usuários.
			\begin{itemize}
				\justifying
				\item<2> São muitas as abordagens possíveis. Porém, todas aplicam alguma variação das seguintes diretrizes básicas: (1) reuniões com \textit{stakeholders}; (2) estabelecimento de regras para preparação e participação; (3) pauta guia para cobrir todos os pontos importantes, com abertura para o estímulo de novas ideias; (4) a reunião é dirigida por uma pessoa específica; (5) é utilizado algum mecanismo de definições e anotação.
			\end{itemize}
		\item<1> \textbf{Requisitos não-funcionais}: atributos de qualidade, desempenho, segurança ou restrição.
	\end{itemize}
\end{frame}

\begin{frame}{Levantamento de requisitos - revisão}
	\begin{itemize}
		\justifying
		\item Sommerville descreve a elicitação como um ciclo de 4 passos:

\begin{frame}{Levantamento de requisitos - revisão}
	\begin{itemize}

		\item<1-3> Sommerville também cita duas técnicas principais:
			\begin{itemize}
				\item<1-3> Entrevista

				\item<4> \textbf{Validação de requisitos}: exame dos requisitos em termos de inconsistência, omissão e ambiguidade.
			\end{itemize}
	\end{itemize}
\end{frame}

\subsubsection{Modelagem - revisão}

\begin{frame}{Modelagem - revisão}
	\begin{itemize}
		\justifying
		\item A atividade de modelagem consiste em criar \textbf{modelos gráficos} de como o sistema deve funcionar.
		\item Cada atividade de modelagem gera um conjunto de modelos específicos, os quais são organizados de acordo com seu tipo, ou propósito.
		\item Os modelos criados consistem em um ou mais diagramas especificados pela \href{https://www.omg.org/spec/UML/2.5.1/PDF}{UML} (\textit{Unified Modeling Language}).
	\end{itemize}
\end{frame}

\begin{frame}{Modelagem - revisão}
	\begin{itemize}
		\justifying
		\item Não existe um padrão único de modelos. Por exemplo:
			\begin{itemize}
				\justifying
				\item Pressman \& Maxim (capítulo 8): Modelagem de requisitos
					\begin{itemize}
						\item Modelagem baseada em cenários;
						\item Modelagem baseada em classes;
						\item Modelagem funcional;
						\item Modelagem comportamental;
					\end{itemize}
				\item Sommerville (capítulo 5): Modelagem de sistemas
					\begin{itemize}
						\item Modelos de contexto;
						\item Modelos de interação;
						\item Modelos estruturais;
						\item Modelos comportamentais;
					\end{itemize}
			\end{itemize}
	\end{itemize}
\end{frame}

\subsubsection{Padrões de arquitetura - revisão}

\begin{frame}{Padrões de arquitetura - revisão}
	\begin{itemize}
		\justifying
		\item Um padrão de arquitetura (ou padrão arquitetural) é uma solução já estabelecida para desenvolvimento de softwares, sendo um modelo reutilizável para problemas já conhecidos. Esses padrões já foram estudados, testados em projetos reais e passaram por melhorias, ou seja, são soluções difundidas e aceitas no mercado\footnote{Fonte: \href{https://www.alura.com.br/artigos/padroes-arquiteturais-arquitetura-software-descomplicada}{Alura}.}.
		\item O padrão a ser utilizado ``desenvolvido'' junto com a modelagem.
	\end{itemize}
\end{frame}

\begin{frame}{Padrões de arquitetura - revisão}
	\begin{itemize}
		\justifying
		\item Alguns dos padrões mais conhecidos e utilizados:
			\begin{itemize}
				\item Arquitetura monolítica.
				\item Arquitetura cliente-servidor.
				\item Arquitetura em camadas (\textit{layers}).
				\item Arquitetura MVC (\textit{Model-View-Controller}).
				\item Arquitetura baseada em microsserviços.
			\end{itemize}
	\end{itemize}
\end{frame}

\subsection{Documentação de processo}

\begin{frame}{Documentação de processo}
	\begin{itemize}
		\justifying
		\item Nesta parte do documento deverão estar presentes todos os registros referentes ao processo de desenvolvimento.
		\item Por exemplo, supondo o Scrum, no documento podem estar presentes as atas ou resumos de \textit{sprints}, \textit{backlogs}, etc.
	\end{itemize}
\end{frame}

% SUBSECTION 1.3 ---------------------------------------------------------------
%-------------------------------------------------------------------------------
\subsection{Registro de decisões}

\begin{frame}{Registro de decisões}
	\begin{itemize}
		\justifying
		\item São registros de alto valor histórico e encapsulam as razões por detrás de cada decisão significativa tomadas durante o projeto.
		\item Assim como as demais partes da documentação, esses registros também devem ser compreensíveis e de fácil acessibilidade.
	\end{itemize}
\end{frame}
