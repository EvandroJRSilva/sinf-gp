\section{Testes}

\begin{frame}{Testes}
	\begin{itemize}
		\justifying
		\item Parte da garantia da qualidade se dá com a realização de testes de software.
		\item Em geral existem quatro níveis de teste:
			\begin{enumerate}
				\item Teste de unidade (\textit{Unity Test})
				\item Teste de integração (\textit{Integration Testing})
				\item Teste de sistema (\textit{System Testing})
				\item Teste de aceitação (\textit{Acceptance Testing})
			\end{enumerate}
	\end{itemize}
\end{frame}

\begin{frame}{Testes}
	\begin{itemize}
		\justifying
		\item Teste de unidade (\textit{Unity Test})
			\begin{itemize}
				\justifying
				\item Teste sobre funções ou componentes individuais.
				\item Garante o comportamento correto no menor nível.
				\item Serve para encontrar e corrigir os primeiros erros.
				\item Técnicas (não se resumem somente ao teste de unidade)
					\begin{itemize}
						\justifying
						\item \textbf{Black box}: baseado nas saídas esperadas para as respectivas entradas, sem o conhecimento da lógica interna ou do código. Foca sobre a validação de funcionalidades sob a perspectiva do usuário, e que o sistema se comporta corretamente de acordo com as entradas fornecidas.
						\item \textbf{White box}: a estrutura interna, o código e a lógica são conhecidos. Garante que o fluxo de dados, condições, etc., estão funcionando como esperado.
						\item \textbf{Gray box}: combinação das técnicas anteriores.
					\end{itemize}
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Testes}
	\begin{itemize}
		\justifying
		\item Teste de integração (\textit{Integration Testing})
			\begin{itemize}
				\item Garante o fluxo correto dos dados entre os módulos.
				\item Serve para encontrar e corrigir os erros que ocorrem durante a interação entre partes diferentes do sistema.
				\item Técnicas
					\begin{itemize}
						\justifying
						\item \textbf{Big-Bang}.
						\item \textbf{Bottom-Up}.
						\item \textbf{Top-Down}.
						\item \textbf{Mixed}.
					\end{itemize}
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Testes}
	\begin{itemize}
		\justifying
		\item Teste de sistema (\textit{System Testing})
		\begin{itemize}
			\item Avalia o sistema totalmente integrado para garantir que o software alcance os requisitos especificados.
			\item Checa se o sistema inteiro funciona como esperado em um ambiente de mundo real.
			\item Processo
			\begin{enumerate}
				\justifying
				\item Configuração do ambiente de teste.
				\item Criação de casos de teste.
				\item Criação de dados de teste.
				\item Execução dos casos teste.
				\item Relatório de defeitos.
				\item Teste de regressão (teste sobre os efeitos colaterais do processo de teste).
				\item Correção de defeitos.
				\item Reteste, se necessário.
			\end{enumerate}
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Testes}
	\begin{itemize}
		\justifying
		\item Teste de aceitação (\textit{Acceptance Testing})
		\begin{itemize}
			\item O(s) cliente(s) verifica(m) se o sistema atende às suas necessidades e expectativas.
			\item Tipos de teste
			\begin{itemize}
				\justifying
				\item \textit{User Acceptance Testing} (UAT).
				\item \textit{Business Acceptance Testing} (BAT).
				\item \textit{Contract Acceptance Testing} (CAT).
				\item \textit{Regulations Acceptance Testing} (RAT).
				\item \textit{Operational Acceptance Testing} (OAT).
				\item \textit{Alpha Testing}.
				\item \textit{Beta Testing}.
			\end{itemize}
		\end{itemize}
	\end{itemize}
\end{frame}

\section{Métricas e Análise de Software}

\begin{frame}{Métricas e Análise de Software}
	\begin{itemize}
		\justifying
		\item Medições podem ser aplicadas ao processo de software com a intenção de melhorá-lo continuamente.
		\item Elas podem ser usadas durante um projeto de software para ajudar nas \textbf{estimativas}, no \textbf{controle de qualidade}, na \textbf{produtividade} e no \textbf{controle de projeto}.
	\end{itemize}
\end{frame}

\begin{frame}{Métricas e Análise de Software}
	\begin{itemize}
		\justifying
		\item No contexto do processo de software uma equipe de software se preocupa principalmente com as \textbf{métricas de produtividade e qualidade} --- medidas da ``saída'' do desenvolvimento de software em função do esforço e do tempo aplicados e medidas da ``adequação para uso'' dos artefatos produzidos.
	\end{itemize}
\end{frame}

\subsection{Medidas, métricas e indicadores}

\begin{frame}{Medidas, métricas e indicadores}
	\begin{itemize}
		\justifying
		\item<1-4> Embora os termos \alert<2>{\textbf{medida}}, \alert<3>{\textbf{medição}} e \alert<4>{\textbf{métricas}} sejam, com frequência, usados indistintamente, é importante notar as diferenças sutis entre eles.
			\begin{itemize}
				\justifying
				\item<2> Quando um único dado é coletado (p. ex., o número de erros descobertos em um componente de software).
				\item<3> Ocorre como resultado da coleta de um ou mais pontos de dados (p. ex., um conjunto de revisões de componente e testes de unidade são investigados para coletar medidas do número de erros para cada um).
				\item<4> Relaciona as medidas individuais de alguma maneira (p. ex., o número médio de erros encontrados por revisão ou o número médio de erros encontrados por teste de unidade).
			\end{itemize}
		\item<5-> Um engenheiro de software coleta medidas e desenvolve métricas para obter \alert<6>{\textbf{indicadores}}.
			\begin{itemize}
				\justifying
				\item<6> É uma métrica ou combinação de métricas que fornecem informações sobre o processo de software, em um projeto de software ou no próprio produto.
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Medidas, métricas e indicadores}
	\begin{itemize}
		\justifying
		\item Classificação das métricas (de acordo com Pressman \& Maxim --- Capítulo 23):
			\begin{itemize}
				\item Métricas de produto
					\begin{itemize}
						\justifying
						\item Métricas para o modelo de requisitos
						\item Métricas de projeto para software convencional
						\item Métricas de projeto para software orientado a objetos
						\item Métricas de projeto de interface de usuário
						\item Métricas para código-fonte
					\end{itemize}
				\item Métricas para teste
				\item Métricas para manutenção
				\item Métricas de processo e de projeto
				\item Medição de software
				\item Métricas para qualidade de software
			\end{itemize}
	\end{itemize}
\end{frame}

\subsection{Análise de dados de software}

\begin{frame}{Análise de dados de software}
	\begin{itemize}
		\justifying
		\item As métricas são usadas para avaliar a qualidade ou o desempenho de um produto ou processo.
		\item Os \textbf{indicadores-chave de desempenho} (KPI - \textit{Key Performance Indicators}) são métricas usadas para acompanhar o desempenho e ativar ações compensatórias quando seus valores cam abaixo de uma faixa predeterminada.
		\item Mas como saber se as métricas são mesmo significativas?
	\end{itemize}
\end{frame}

\begin{frame}{Análise de dados de software}
	\begin{itemize}
		\justifying
		\item<1> A \textbf{análise de software} (\textit{software analytics}) é a análise computacional sistemática de dados ou estatísticas de engenharia de software para fornecer aos gerentes e engenheiros de software ideias e observações significativas e capacitar suas equipes para a tomada de melhores decisões.
		\item<2> Por exemplo: saber o número de defeitos em um artefato de software hoje é menos relevante do que saber que o número de defeitos é 5\% maior do que era no mês passado.
			\begin{itemize}
				\justifying
				\item<2> A análise pode ajudar os desenvolvedores a prever o número esperado de defeitos, onde realizar testes para descobri-los e quanto tempo demorará para consertá-los.
			\end{itemize}
	\end{itemize}
\end{frame}
