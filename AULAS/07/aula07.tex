\section{Gerenciamento de Pessoas}

\setbeamertemplate{background canvas}{%
    \tikz[remember picture,overlay] \node[opacity=0.15,inner sep=0pt] at (current page.center){\includegraphics[width=\paperwidth,height=\paperheight]{../backgrounds/developer02.jpg}};
}

\begin{frame}{Gerenciamento de Pessoas}
    \begin{itemize}
        \justifying
        \item As pessoas que trabalham em uma organização de software são os seus \textbf{maiores ativos}
        	\begin{itemize}
        		\justifying
        		\item<1-2> É caro recrutar e reter pessoas boas.
        		\item<2> Cabe aos gerentes de software garantir que os engenheiros que trabalham em um projeto sejam tão produtivos quanto for possível.
        	\end{itemize}
        \item<3> Essa produtividade é alcançada quando as pessoas são \textbf{motivadas} e  \textbf{respeitadas} pela organização, e \textbf{recebem responsabilidades que refletem suas habilidades e experiência}.
    \end{itemize}
\end{frame}

\begin{frame}{Gerenciamento de Pessoas}
	\begin{itemize}
		\justifying
		\item<1> Além de \textbf{compreender as questões técnicas} que influenciam o desenvolvimento de software, o gerente de projetos deve estar ciente dos possíveis problemas de gerenciamento de pessoas e \textbf{tentar desenvolver habilidades para gerenciar pessoas} (\textit{soft skills}).
		\item<2-> Existem quatro fatores críticos que influenciam o relacionamento entre um gerente e as pessoas que ele gerencia:
			\begin{enumerate}
				\justifying
				\item<3> \textbf{Consistência}. Todas as pessoas em um time de projeto devem ser tratadas de modo comparável.
				\item<4> \textbf{Respeito}. Pessoas diferentes têm habilidades diferentes, e os gerentes devem respeitar essas diferenças.
				\item<5> \textbf{Inclusão}. As pessoas contribuem efetivamente quando sentem que são ouvidas e que suas propostas são levadas em consideração.
				\item<6> \textbf{Honestidade}. O gerente deve sempre ser honesto a respeito do que está indo bem e do que está indo mal com o time; também deve \textbf{reconhecer seu próprio nível de conhecimento técnico} e \textbf{estar disposto a submeter-se à pessoa com mais conhecimento quando for necessário}.
			\end{enumerate}
	\end{itemize}
\end{frame}

\subsection{Motivação das pessoas}

\begin{frame}{Motivação das pessoas}
	\begin{itemize}
		\justifying
		\item O gerente de projetos deve \textbf{motivar} as pessoas do time \textbf{para que elas contribuam com o melhor das habilidades delas}.
			\begin{itemize}
				\justifying
				\item \textbf{Se as pessoas não estiverem motivadas}, elas estarão menos interessadas no trabalho que estão fazendo e \textbf{trabalharão mais devagar}, estarão \textbf{mais propensas a cometer erros} e \textbf{não contribuirão para as metas} do time ou da organização.
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Motivação das pessoas}
	\begin{columns}
		\begin{column}{0.4\textwidth}
			\centering
			\textbf{Modelo de Maslow}\\
			\vspace{5pt}
			\includegraphics[width=1.3\textwidth]{imagens/figura01.png}
		\end{column}
		
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\small
				\justifying
				\item<1> Para satisfazer as \textbf{necessidades sociais}: dar tempo às pessoas para se encontrarem com seus colegas de trabalho e proporcionar lugares para que esses encontros aconteçam.
				\item<2> Para satisfazer as \textbf{necessidades de estima}: mostrar às pessoas que elas são valorizadas pela organização.
				\item<3> Para satisfazer as \textbf{necessidades de autorrealização}: dar às pessoas responsabilidade, alocá-las em tarefas que exijam delas e proporcionar oportunidades para treinamento e desenvolvimento.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Motivação das pessoas}
	\begin{itemize}
		\justifying
		\item O modelo de motivação de Maslow é útil até certo ponto, mas ele não leva em conta o fato de que as \textbf{pessoas se sentem parte de uma organização}, um grupo profissional e uma ou mais culturas.
		\item \textbf{Ser membro de um grupo coeso é altamente motivador para a maioria das pessoas}. As pessoas com trabalhos gratificantes costumam gostar de ir trabalhar porque são motivadas pelas pessoas com as quais trabalham e pelo trabalho que realizam
		\item Portanto, \textbf{o gerente também deve pensar em como um grupo pode ser motivado como um todo}.
	\end{itemize}
\end{frame}

\section{Trabalho em Equipe}

\setbeamertemplate{background canvas}{%

\begin{frame}{Trabalho em Equipe}
	\begin{itemize}
		\justifying
		\item<1> Em \textbf{equipes pequenas} há \textbf{menos problemas de comunicação}
		\begin{itemize}
			\justifying
			\item<1> Todos se conhecem, e o grupo todo pode sentar em uma mesa para se reunir e discutir sobre o projeto e o software que estão desenvolvendo.
		\end{itemize}
		\item<2-> Já as \textbf{grandes equipes} precisam ser divididas em \textbf{grupos menores}
		\begin{itemize}
			\justifying
			\item<2> Cada grupo fica responsável por uma parte do software.
			\item<2> \textbf{Desafio}: equilibrar as habilidades técnicas, experiências e personalidades $\rightarrow$ grupo coeso.
			\item<3> ``O melhor tamanho para um grupo de engenharia de software é de quatro a seis membros e eles nunca devem ter mais de 12 membros'' (Sommerville).
		\end{itemize}
	\end{itemize}
\end{frame}

\subsection{Grupo Coeso}

\setbeamertemplate{background canvas}{%
	\tikz[remember picture,overlay] \node[opacity=0.15,inner sep=0pt] at (current page.center){\includegraphics[width=\paperwidth,height=\paperheight]{../backgrounds/developer03.jpg}};
}

\begin{frame}{Grupo Coeso}
	\begin{itemize}
		\justifying
		\item Seus membros
		\begin{itemize}
			\justifying
			\item<1-> São \textbf{motivados pelo sucesso do grupo} e também, por seus próprios objetivos pessoais.
			\item<2-> São \textbf{leais} e \textbf{consideram o grupo mais importante} do que os indivíduos que o compõem.
			\item<3-> \textbf{Se identificam com os objetivos do grupo} e com os demais membros.
			\item<4> \textbf{Tentam proteger o grupo}, como uma entidade, de interferências externas.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Grupo Coeso}
	\begin{itemize}
		\justifying
		\item Benefícios de um grupo coeso no desenvolvimento de software
		\begin{enumerate}
			\justifying
			\item<1> \textbf{O grupo pode estabelecer seus próprios padrões de qualidade}: Como esses padrões são estabelecidos por consenso, eles estão mais propensos a serem seguidos do que os padrões externos impostos ao grupo.
			\item<2> \textbf{Indivíduos aprendem e apoiam uns aos outros}: Os membros do grupo aprendem trabalhando juntos. As inibições causadas pela ignorância são minimizadas à medida que a aprendizagem mútua é incentivada.
		\end{enumerate}
	\end{itemize}
\end{frame}

\begin{frame}{Grupo Coeso}
	\begin{itemize}
		\justifying
		\item Benefícios de um grupo coeso no desenvolvimento de software
		\begin{enumerate}
			\justifying
			\setcounter{enumi}{2}
			\item<1> \textbf{O conhecimento é compartilhado}: A continuidade pode ser mantida se um membro do grupo sair. Outros no grupo podem assumir tarefas criticas e assegurar que o projeto não seja indevidamente interrompido.
			\item<2> \textbf{A refatoração e a melhoria continua são incentivadas}: Os membros do grupo trabalham coletivamente para fornecer resultados de alta qualidade e corrigir problemas, independentemente dos indivíduos que criaram o projeto (design) ou programa originalmente.
		\end{enumerate}
	\end{itemize}
\end{frame}

\begin{frame}{Grupo Coeso}
	\begin{itemize}

		\item \textbf{Bons gerentes de projetos devem sempre tentar incentivar a coesão do grupo}

\subsection{Organização do grupo}

\setbeamertemplate{background canvas}{%
	\tikz[remember picture,overlay] \node[opacity=0.15,inner sep=0pt] at (current page.center){\includegraphics[width=\paperwidth,height=\paperheight]{../backgrounds/Gemini_Generated_Image_8gu9xk8gu9xk8gu9.png}};
}

\begin{frame}{Organização do grupo}
	%\centering
	\begin{tblr}{colspec={Q[0.48\textwidth] Q[0.48\textwidth]}, hlines, vlines, row{1}={c}, hline{1,2}={1.5pt, solid}, vline{1-3}={1}{1.5pt, solid}}
		\textbf{Grupos hierárquicos} & \textbf{Grupos pequenos}\\
		\justifying\scriptsize O líder do grupo está no topo da hierarquia, tem autoridade mais formal do que os membros do grupo e, portanto, pode dirigir o trabalho deles. & \justifying\scriptsize Normalmente são organizados de maneira informal e o líder se envolve no desenvolvimento de software com outros membros do grupo.\\
		\justifying\scriptsize Há uma estrutura organizacional clara e as decisões são tomadas na direção do topo da hierarquia e implementadas pelas pessoas mais abaixo. & \justifying\scriptsize Todos discutem o trabalho a ser executado, e as tarefas são alocadas de acordo com a capacidade e a experiência.\\
		\justifying\scriptsize Conseguem trabalhar bem quando um problema bem compreendido pode ser facilmente dividido em componentes de software que possam ser desenvolvidos em diferentes partes da hierarquia. & \justifying\scriptsize Podem ser bem sucedidos quando a maioria dos membros do grupo é experiente e competente. Se um grupo for composto principalmente de membros pouco experientes ou pouco competentes o resultado pode ser a falta de coordenação entre os membros do grupo e, possivelmente, o eventual fracasso do projeto.
	\end{tblr}
\end{frame}

\begin{frame}{Organização do grupo}
	\begin{itemize}
		\justifying
		\item As \textbf{principais responsabilidades} do gerente incluem:
			\begin{itemize}
				\justifying
				\item Determinar os cargos necessários.
				\item Definir quem melhor se encaixa em cada cargo.
				\item Estabelecer canais eficientes de comunicação.
			\end{itemize}
		\item<2-> \textbf{Desafio}: alinhar o design do time com as demandas do projeto enquanto balanceia talentos e proficiências individuais.
		\item<3> \textbf{Para ter sucesso} o gerente precisa \textbf{ter um conhecimento robusto sobre os processos de desenvolvimento}, \textbf{inteligência emocional} (para relacionamentos interpessoais efetivos) e \textbf{habilidade de planejamento estratégico} para organização adaptável dos times.
	\end{itemize}
\end{frame}

\subsection{Comunicação}

\setbeamertemplate{background canvas}{%
	\tikz[remember picture,overlay] \node[opacity=0.15,inner sep=0pt] at (current page.center){\includegraphics[width=\paperwidth,height=\paperheight]{../backgrounds/Gemini_Generated_Image_6b9mzp6b9mzp6b9m.png}};
}

\begin{frame}{Comunicação}
	\begin{itemize}
		\justifying
		\item É \textbf{absolutamente essencial} que os membros do grupo se comuniquem com \textbf{eficácia} e \textbf{eficiência} uns com os outros e com os outros \textit{stakeholders} do projeto.
		\item Os membros do grupo devem trocar informações sobre
			\begin{itemize}
				\justifying
				\item O status do seu trabalho.
				\item As decisões de projeto que foram tomadas.
				\item As mudanças nas decisões de projeto anteriores.
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Comunicação}
	\begin{itemize}
		\justifying
		\item<1> A eficácia e a eficiência da comunicação são influenciadas por:
			\begin{enumerate}
				\item<1> Tamanho do grupo.
				\item<1> Estrutura do grupo.
				\item<1> Composição do grupo.
				\item<1> Ambiente físico de trabalho.
				\item<1> Canais de comunicação disponíveis.
			\end{enumerate}
		\item<2> A comunicação eficaz é alcançada quando é \textbf{bidirecional} e as pessoas envolvidas podem discutir questões e informações e estabelecer um entendimento comum das propostas e problemas.
			\begin{itemize}
				\justifying
				\item<2> Isso pode ser feito por meio de \textbf{reuniões}.
				\item<2> Mas pode ser feito também através de \textbf{discussões informais}, nas quais um gerente se reúne com o time para um café (muitas vezes mais eficazes).
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Comunicação}
	\begin{itemize}
		\justifying
		\item Cada vez mais times de projeto incluem membros remotos, o que também torna mais difícil fazer reuniões [presenciais]. ara envolvê-los na comunicação, também é possível usar:
			\begin{itemize}
				\justifying
				\item \textbf{Wikis}: apoiam a criação colaborativa e a edição de documentos.
				\item \textbf{Blogs}: apoiam discussões encadeadas sobre perguntas e comentários feitos pelos membros do grupo.
				\item \textbf{Aplicativos mensageiros}: WhatsApp ou Telegram.
				\item \textbf{Teleconferências}: Google Meet, MS Teams, etc.
			\end{itemize}
	\end{itemize}
\end{frame}

\subsection{Avaliação de desempenho}

\setbeamertemplate{background canvas}{%
	\tikz[remember picture,overlay] \node[opacity=0.15,inner sep=0pt] at (current page.center){\includegraphics[width=\paperwidth,height=\paperheight]{../backgrounds/Gemini_Generated_Image_fhwp3yfhwp3yfhwp.png}};
}


\begin{frame}{Avaliação de desempenho}
	\begin{itemize}
		\justifying
		\item A avaliação de desempenho consiste em fornecer \textbf{regularmente} \textbf{\textit{feedbacks} construtivos} aos membros equipe.
		\item Uma responsabilidade fundamental é avaliar o desempenho dos membros da equipe no cumprimento de suas metas de entrega e na contribuição para os projetos. 
		\item É crucial \textbf{definir métricas} de resultados \textbf{claras} e \textbf{manter um diálogo constante sobre o progresso}.
	\end{itemize}
\end{frame}

\begin{frame}{Avaliação de desempenho}
	\begin{itemize}
		\justifying
		\item \textbf{Desafio}
			\begin{itemize}
				\justifying
				\item Encontrar um ponto de equilíbrio entre a crítica e o reconhecimento.
				\item É essencial manter perspectiva justa e sem enviesamento, e comunicar o \textit{feedback} de forma construtiva.
			\end{itemize}
		\item Para ter sucesso um Gerente precisa ter uma \textbf{forte habilidade em comunicação}, \textbf{empatia}, e um foco maior sobre resolução de problemas, em vez de ficar procurando falhas.
		\item Ao promover um ambiente aberto e transparente onde as discussões relacionadas ao desempenho são incentivadas, os Gerentes podem garantir o desenvolvimento e o crescimento consistentes dentro da equipe.
	\end{itemize}
\end{frame}

\subsection{Planejamento do desenvolvimento de carreira}

\begin{frame}{Planejamento do desenvolvimento de carreira}
	\begin{itemize}
		\justifying
		\item O suporte do Gerente ao crescimento na carreira de seu time é imprescindível. Isso requer um \textbf{alinhamento de crescimento individual} com os \textbf{objetivos da empresa} e as \textbf{necessidades de habilidades do time}.
		\item Não consiste apenas em discutir sobre cargos futuros, mas também em promover um \textbf{crescimento profissional e aprendizado em longo prazo}.
	\end{itemize}
\end{frame}

\begin{frame}{Planejamento do desenvolvimento de carreira}
	\begin{itemize}
		\justifying
		\item \textbf{Desafio}
		\begin{itemize}
			\justifying
			\item Encontrar um ponto de equilíbrio entre a carga de trabalho atual e as necessidades de desenvolvimento de cada membro.
		\end{itemize}
		\item Para ter sucesso, um gerente precisa ter \textbf{fortes habilidades de comunica-ção}, \textbf{empatia} e \textbf{compromisso com o desenvolvimento do time}. 
		\item Um gerente eficaz precisa estar apto a \textbf{guiar o seu time} e garantir que \textbf{estejam com sua carreira em crescimento}, ao mesmo tempo em que \textbf{atingem as metas da empresa}.
	\end{itemize}
\end{frame}

\subsection{Mentoria e treinamento (\textit{coaching})}

\setbeamertemplate{background canvas}{%
	\tikz[remember picture,overlay] \node[opacity=0.15,inner sep=0pt] at (current page.center){\includegraphics[width=\paperwidth,height=\paperheight]{../backgrounds/Gemini_Generated_Image_8gu9xk8gu9xk8gu9.png}};
}

\begin{frame}{Mentoria e treinamento (\textit{coaching})}
	\begin{itemize}
		\justifying
		\item Um Gerente de Projetos frequentemente de mentoria e treinamento de seu time. Ele é responsável por \textbf{fornecer \textit{feedbacks} regulares}, \textbf{aconselhando sobre} o \textbf{desenvolvimento técnico e profissional}, e também sobre \textbf{quais objetivos traçar}.
		\item Esse envolvimento ajuda a cultivar uma cultura de aprendizado e crescimento contínuos.
	\end{itemize}
\end{frame}

\begin{frame}{Mentoria e treinamento (\textit{coaching})}
	\begin{itemize}
		\justifying
		\item \textbf{Desafio}
			\begin{itemize}
				\justifying
				\item Encontrar o ponto certo entre dar suporte ou incentivar os membros da equipe a encontrarem uma solução por conta própria.
				\item Manter a imparcialidade e a consistência na abordagem que adotam em relação a diferentes indivíduos.
			\end{itemize}
		\item Para ter sucesso um Engenheiro precisa de forte habilidade de comunicação, empatia e um bom entendimento de pontos fortes e fracos de cada membro de sua equipe.
		\item Além disso, é adotar um pensamento mais voltado ao treinamento, de forma ajudar sua equipe a sobrepor obstáculos, desenvolver novas habilidades e alcançar seu potencial completo.
	\end{itemize}
\end{frame}
