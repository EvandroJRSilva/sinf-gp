\begin{frame}{Comunicação}
    \begin{itemize}
        \justifying
        \item Há muitas razões para que o projeto de software tenha problemas
        	\begin{itemize}
        		\justifying
        		\item Escala, retrocompatibilidade e complexidade.
        		\item Confusão e dificuldades de coordenação de membros.
        		\item Incertezas e mudanças contínuas.
        		\item etc.
        	\end{itemize}
        \item<2> Para lidar com elas, devem-se estabelecer métodos eficazes para coordenar as pessoas que realizam o trabalho. Para tanto, devem-se estabelecer \textbf{mecanismos para comunicação formal e informal} entre os membros da equipe.
    \end{itemize}
\end{frame}

\begin{frame}{Comunicação}
	\begin{itemize}
		\justifying
		\item \textbf{Reuniões pessoais}
			\begin{itemize}
				\justifying
				\item Apenas o gerente e um membro da equipe.
				\item O gerente deve ser capaz de entender as preocupações do membro de seu time, dar feedback valioso e estabelecer trajetórias de crescimento individual.
				\item \textbf{Desafio}: fornecer feedback negativo sem desmoralizar o funcionário.
				\item \textbf{Sucesso}: requer habilidades de escuta, empatia, mentalidade voltada à solução de problemas e comunicação efetiva. O gerente precisa ser acessível e dar segurança para seus funcionários expressarem problemas e receios.
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Comunicação}
	\begin{itemize}
		\justifying
		\item \textbf{Reuniões do time}
		\begin{itemize}
			\justifying
			\item O gerente deve guiar a reunião, elencando os pontos-chave a serem discutidos. Deve também utilizar comunicação clara e ouvir ideias e feedbacks do time.
			\item \textbf{Desafio}: garantir que todos serão ouvidos e que não haja perda de tempo.
			\item \textbf{Sucesso}: requer habilidades de organização e interpessoais, além de avaliar perspectivas diferentes e promover uma cultura de respeito e mente aberta.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Comunicação}
	\begin{itemize}
		\justifying
		\item \textbf{Relatório de Status}
		\begin{itemize}
			\justifying
			\item Envolve o monitoramento constante dos projetos, lidar com gargalos e atualizar a alta gerência e \textit{stakeholders}. O Gerente deve garantir que todos estejam informados sobre os cronogramas do projeto, alocação de recursos e progresso do desenvolvimento.
			\item \textbf{Desafio}: entregar más notícias ``diplomaticamente'' (alterações no cronograma, estouros de orçamento ou desafios técnicos).
			\item \textbf{Sucesso}: requer ter uma visão clara de todos os status do projeto e estar pronto a fornecer atualizações concisas e acuradas.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Comunicação}
	\begin{itemize}
		\justifying
		\item \textbf{Gerenciamento de \textit{Stakeholder}}
		\begin{itemize}
			\justifying
			\item O Gerente deve ser capaz de esclarecer os objetivos do projeto, lidar com problemas e criar uma visão compartilhada entre as diferentes partes interessadas. As principais responsabilidades incluem conduzir reuniões, gerenciar expectativas e fornecer atualizações regulares sobre o progresso.
			\item \textbf{Desafio}: balancear visões opostas e alcançar um consenso.
			\item \textbf{Sucesso}: requer ter um entendimento das dinâmicas únicas do projeto e das tecnologias, além de fortes habilidades de comunicação para \textbf{transmitir informações técnicas de forma não técnica}.
		\end{itemize}
	\end{itemize}
\end{frame}
